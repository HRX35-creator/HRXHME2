cmake_minimum_required(VERSION 3.20)
project(seekdb_hipvs_build LANGUAGES C)
set(HIPVS_ROOT "" CACHE PATH "Installed hipVS SDK prefix")
set(ROCM_ROOT "/opt/rocm" CACHE PATH "Installed ROCm SDK prefix")
find_path(HIPVS_INCLUDE cuvs/neighbors/cagra.h HINTS "${HIPVS_ROOT}/include" REQUIRED)
find_path(ROCM_INCLUDE hip/hip_runtime.h HINTS "${ROCM_ROOT}/include" REQUIRED)
find_path(DLPACK_INCLUDE dlpack/dlpack.h HINTS "${HIPVS_ROOT}/include" "$ENV{CONDA_PREFIX}/include" REQUIRED)
find_library(HIPVS_C_LIBRARY NAMES cuvs_c hipvs_c HINTS "${HIPVS_ROOT}/lib" "${HIPVS_ROOT}/lib64" REQUIRED)
add_library(seekdb_hipvs_build SHARED seekdb_hipvs_build.c)
target_compile_features(seekdb_hipvs_build PRIVATE c_std_11)
target_compile_definitions(seekdb_hipvs_build PRIVATE __HIP_PLATFORM_AMD__)
target_include_directories(seekdb_hipvs_build PRIVATE "${HIPVS_INCLUDE}" "${ROCM_INCLUDE}" "${DLPACK_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../src/oblib")
target_link_libraries(seekdb_hipvs_build PRIVATE "${HIPVS_C_LIBRARY}")
target_link_options(seekdb_hipvs_build PRIVATE "LINKER:--no-undefined")
install(TARGETS seekdb_hipvs_build LIBRARY DESTINATION lib)
