# Standalone compatibility test against the same VSAG SDK as the server.
# No GPU or hipVS is needed for the format/consumer regression.
cmake_minimum_required(VERSION 3.20)
project(hipvs_graph_import_test LANGUAGES CXX)
enable_testing()
set(VSAG_ROOT "" CACHE PATH "Root containing include/vsag and lib/vsag_lib")
find_path(VSAG_INCLUDE vsag/vsag.h HINTS "${VSAG_ROOT}/include" REQUIRED)
find_library(VSAG_LIBRARY vsag HINTS "${VSAG_ROOT}/lib/vsag_lib" "${VSAG_ROOT}/lib" REQUIRED)
add_executable(test_hipvs_graph_import test_hipvs_graph_import.cpp)
target_compile_features(test_hipvs_graph_import PRIVATE cxx_std_20)
target_include_directories(test_hipvs_graph_import PRIVATE "${VSAG_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/oblib")
target_link_libraries(test_hipvs_graph_import PRIVATE "${VSAG_LIBRARY}" pthread)
add_test(NAME hipvs_graph_import COMMAND test_hipvs_graph_import)
add_executable(replay_hipvs_graph replay_hipvs_graph.cpp)
target_compile_features(replay_hipvs_graph PRIVATE cxx_std_20)
target_include_directories(replay_hipvs_graph PRIVATE "${VSAG_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/oblib")
target_link_libraries(replay_hipvs_graph PRIVATE "${VSAG_LIBRARY}" pthread)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(fake_hipvs_build SHARED fake_hipvs_build.cpp)
  set_target_properties(fake_hipvs_build PROPERTIES OUTPUT_NAME seekdb_hipvs_build LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fake")
  target_include_directories(fake_hipvs_build PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/oblib")
  target_link_libraries(fake_hipvs_build PRIVATE pthread)
  add_executable(test_hipvs_build_loader test_hipvs_build_loader.cpp)
  target_compile_features(test_hipvs_build_loader PRIVATE cxx_std_20)
  target_include_directories(test_hipvs_build_loader PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/oblib")
  target_link_libraries(test_hipvs_build_loader PRIVATE dl pthread)
  add_test(NAME hipvs_loader_missing COMMAND test_hipvs_build_loader)
  set_tests_properties(hipvs_loader_missing PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=")
  add_test(NAME hipvs_loader_faults COMMAND test_hipvs_build_loader --fake)
  set_tests_properties(hipvs_loader_faults PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}/fake")
endif()
